    public void insert(
            Long parentUserId,
            Long childUserId,
            String childName,
            Long firstOptionId,
            Long secondOptionId,
            Long thirdOptionId,
            String comment) {

        EntityManager em = null;

        try {
            em = JpaUtil.createEntityManager();
            em.getTransaction().begin();

            Query requestQuery =
                    em.createNativeQuery(
                            "INSERT INTO parent_requests "
                                    + "(parent_user_id, "
                                    + "child_user_id, "
                                    + "child_name, "
                                    + "comment, "
                                    + "request_status) "
                                    + "VALUES "
                                    + "(:parentUserId, "
                                    + ":childUserId, "
                                    + ":childName, "
                                    + ":comment, "
                                    + "'SUBMITTED')"
                    );

            requestQuery.setParameter(
                    "parentUserId",
                    parentUserId
            );

            requestQuery.setParameter(
                    "childUserId",
                    childUserId
            );

            requestQuery.setParameter(
                    "childName",
                    childName
            );

            requestQuery.setParameter(
                    "comment",
                    comment
            );

            requestQuery.executeUpdate();

            Query idQuery =
                    em.createNativeQuery(
                            "SELECT LAST_INSERT_ID()"
                    );

            Long requestId =
                    ((Number) idQuery
                            .getSingleResult())
                            .longValue();

            insertRequestOption(
                    em,
                    requestId,
                    firstOptionId,
                    1
            );

            if (secondOptionId != null) {
                insertRequestOption(
                        em,
                        requestId,
                        secondOptionId,
                        2
                );
            }

            if (thirdOptionId != null) {
                insertRequestOption(
                        em,
                        requestId,
                        thirdOptionId,
                        3
                );
            }

            em.getTransaction().commit();

        } catch (RuntimeException e) {

            if (em != null
                    && em.getTransaction().isActive()) {

                em.getTransaction().rollback();
            }

            throw e;

        } finally {

            if (em != null && em.isOpen()) {
                em.close();
            }
        }
    }

        );

    }

                    "SELECT pr.id, "

                            + "u.display_name, "

                            + "ON pr.parent_user_id = u.id "
                            + "LEFT JOIN parent_children pc "
                            + "ON pr.parent_user_id "
                            + "= pc.parent_user_id "
                            + "AND pr.child_user_id "
                            + "= pc.child_user_id "
                            + "LEFT JOIN final_plans fp "
                            + "ON pr.id = fp.parent_request_id "

            sql +=
                    "ORDER BY "
                            + "pr.parent_user_id ASC, "
                            + "pr.created_at ASC";

                    em.createNativeQuery(sql);

            if (firstOptionId != null) {
                query.setParameter(

                        firstOptionId

    private void attachOptions(
            List<ParentRequest> requests) {

        for (ParentRequest request : requests) {

            if (request.getId() == null) {
                continue;
            }

            request.setFirstOption(
                    findOptionByRequestIdAndPriority(
                            request.getId(),
                            1
                    )
            );

            request.setSecondOption(
                    findOptionByRequestIdAndPriority(
                            request.getId(),
                            2
                    )
            );

            request.setThirdOption(
                    findOptionByRequestIdAndPriority(
                            request.getId(),
                            3
                    )
            );
        }
    }

    private ReservationOption
    findOptionByRequestIdAndPriority(
            Long requestId,
            int priority) {

        EntityManager em = null;

        try {
            em = JpaUtil.createEntityManager();

            Query query =
                    em.createNativeQuery(
                            "SELECT ro.id, "
                                    + "DATE_FORMAT("
                                    + "ro.option_date, '%Y-%m-%d'), "
                                    + "ro.option_time, "
                                    + "ro.available_flag "
                                    + "FROM parent_request_options pro "
                                    + "JOIN reservation_options ro "
                                    + "ON pro.reservation_option_id "
                                    + "= ro.id "
                                    + "WHERE pro.parent_request_id "
                                    + "= :requestId "
                                    + "AND pro.priority = :priority"
                    );

            query.setParameter(
                    "requestId",
                    requestId
            );

            query.setParameter(
                    "priority",
                    priority
            );

            @SuppressWarnings("unchecked")
            List<Object[]> rows =
                    query.getResultList();

            if (rows.isEmpty()) {
                return null;
            }

            Object[] row = rows.get(0);

            ReservationOption option =
                    new ReservationOption();

            option.setId(
                    ((Number) row[0]).longValue()
            );

            option.setDate(
                    (String) row[1]
            );

            option.setTime(
                    (String) row[2]
            );

            option.setAvailable(
                    toBoolean(row[3])
            );

            return option;

        } finally {

            if (em != null && em.isOpen()) {
                em.close();
            }
        }
    }

    /*
     * 教师手动确认画面中使用的推荐候选。
     * 实际的冲突判定仍然只确认第一希望。
     */
    private void applyRecommendedOptions(
            List<ParentRequest> requests) {

        Set<Long> usedOptionIds =
                new HashSet<>();

        for (ParentRequest request : requests) {

            if (request.getId() == null) {
                continue;
            }

            ReservationOption siblingBase =
                    findSiblingBaseOption(
                            requests,
                            request
                    );

            if (request.isHasSibling()
                    && siblingBase != null) {

                ReservationOption nearOption =
                        findNearSiblingOption(
                                request,
                                siblingBase,
                                usedOptionIds
                        );

                if (nearOption != null) {

                    request.setRecommendedOption(
                            nearOption
                    );

                    request.setRecommendedPriorityLabel(
                            getPriorityLabel(
                                    request,
                                    nearOption
                            )
                    );

                    request.setRecommendationMessage(
                            "兄弟調整：同じ日で"
                                    + "近い時間のため推薦"
                    );

                    usedOptionIds.add(
                            nearOption.getId()
                    );

                    continue;
                }
            }

            ReservationOption normalOption =
                    findFirstAvailableOption(
                            request,
                            usedOptionIds
                    );

            if (normalOption != null) {

                request.setRecommendedOption(
                        normalOption
                );

                request.setRecommendedPriorityLabel(
                        getPriorityLabel(
                                request,
                                normalOption
                        )
                );

                if (request.isHasSibling()) {
                    request.setRecommendationMessage(
                            "兄弟あり："
                                    + "通常の希望順位で推薦"
                    );

                } else {
                    request.setRecommendationMessage(
                            "先着順と希望順位により推薦"
                    );
                }

                usedOptionIds.add(
                        normalOption.getId()
                );

            } else {

                request.setRecommendedOption(null);

                request.setRecommendedPriorityLabel(
                        "要調整"
                );

                if (request.isHasSibling()) {
                    request.setRecommendationMessage(
                            "兄弟あり："
                                    + "利用可能な候補がないため要調整"
                    );

                } else {
                    request.setRecommendationMessage(
                            "利用可能な希望がありません"
                    );
                }
            }
        }
    }

    private ReservationOption
    findFirstAvailableOption(
            ParentRequest request,
            Set<Long> usedOptionIds) {

        ReservationOption[] options = {
                request.getFirstOption(),
                request.getSecondOption(),
                request.getThirdOption()
        };

        for (ReservationOption option : options) {

            if (option != null
                    && option.isAvailable()
                    && !usedOptionIds.contains(
                    option.getId())) {

                return option;
            }
        }

        return null;
    }

    private ReservationOption
    findSiblingBaseOption(
            List<ParentRequest> requests,
            ParentRequest current) {

        for (ParentRequest other : requests) {

            if (other.getId().equals(
                    current.getId())) {

                continue;
            }

            if (other.getParentUserId().equals(
                    current.getParentUserId())
                    && other.getRecommendedOption()
                    != null) {

                return other.getRecommendedOption();
            }
        }

        return null;
    }

    private ReservationOption
    findNearSiblingOption(
            ParentRequest request,
            ReservationOption base,
            Set<Long> usedOptionIds) {

        ReservationOption best = null;
        int bestScore = -1;

        ReservationOption[] options = {
                request.getFirstOption(),
                request.getSecondOption(),
                request.getThirdOption()
        };

        for (ReservationOption option : options) {

            if (option == null
                    || !option.isAvailable()
                    || usedOptionIds.contains(
                    option.getId())) {

                continue;
            }

            int score =
                    scoreSiblingOption(
                            option,
                            base
                    );

            if (score > bestScore) {
                bestScore = score;
                best = option;
            }
        }

        return best;
    }

    private int scoreSiblingOption(
            ReservationOption option,
            ReservationOption base) {

        int score = 0;

        if (option.getDate().equals(
                base.getDate())) {

            score += 100;
        }

        int diff =
                Math.abs(
                        getStartMinutes(
                                option.getTime()
                        )
                                - getStartMinutes(
                                base.getTime()
                        )
                );

        if (diff == 40) {
            score += 60;

        } else if (diff == 80) {
            score += 40;

        } else if (diff == 120) {
            score += 20;
        }

        return score;
    }

    private int getStartMinutes(
            String timeRange) {

        if (timeRange == null
                || !timeRange.contains(":")) {

            return 0;
        }

        String start =
                timeRange.split("-")[0];

        String[] parts =
                start.split(":");

        return Integer.parseInt(parts[0])
                * 60
                + Integer.parseInt(parts[1]);
    }

    private String getPriorityLabel(
            ParentRequest request,
            ReservationOption option) {

        if (request.getFirstOption() != null
                && request.getFirstOption()
                .getId()
                .equals(option.getId())) {

            return "第1希望";
        }

        if (request.getSecondOption() != null
                && request.getSecondOption()
                .getId()
                .equals(option.getId())) {

            return "第2希望";
        }

        if (request.getThirdOption() != null
                && request.getThirdOption()
                .getId()
                .equals(option.getId())) {

            return "第3希望";
        }

        return "推薦候補";
    }

    private List<ParentRequest> convertRows(
            List<Object[]> rows) {

        List<ParentRequest> result =
                new ArrayList<>();

        for (Object[] row : rows) {

            ParentRequest request =
                    new ParentRequest();

            if (row[0] != null) {
                request.setId(
                        ((Number) row[0])
                                .longValue()
                );
            }

            if (row[1] != null) {
                request.setParentUserId(
                        ((Number) row[1])
                                .longValue()
                );
            }

            request.setParentUsername(
                    (String) row[2]
            );

            if (row[3] != null) {
                request.setChildUserId(
                        ((Number) row[3])
                                .longValue()
                );
            }

            request.setChildName(
                    (String) row[4]
            );

            request.setComment(
                    (String) row[5]
            );

            request.setRequestStatus(
                    (String) row[6]
            );

            request.setCreatedAt(
                    (String) row[7]
            );

            request.setHasSibling(
                    toBoolean(row[8])
            );

            request.setSiblingGroup(
                    (String) row[9]
            );

            request.setManualRequired(
                    "MANUAL_REQUIRED".equals(
                            request.getRequestStatus()
                    )
            );

            result.add(request);
        }

        return result;
    }

    private boolean toBoolean(Object value) {

        if (value instanceof Boolean) {
            return (Boolean) value;
        }

        if (value instanceof Number) {
            return ((Number) value)
                    .intValue() == 1;
        }

        return false;
    }
